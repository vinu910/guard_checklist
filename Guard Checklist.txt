<script>
  // ====== YAHAN APNA APPS SCRIPT WEB APP URL PASTE KARO ======
  const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbw13UJh_E2U5laAvnqVUqJFngafmQ2d0_xJ9PtXLoaUvFkMyOrqwnSEy4FwFZ0mPBAz/exec";
  // =============================================================

  if (SCRIPT_URL.indexOf("PASTE_YOUR") === 0) {
    document.getElementById('configWarning').style.display = 'block';
  }

  // default entry datetime = now
  const dtField = document.getElementById('entryDateTime');
  const now = new Date();
  now.setMinutes(now.getMinutes() - now.getTimezoneOffset());
  dtField.value = now.toISOString().slice(0,16);

  const photoTrigger = document.getElementById('photoTrigger');
  const photoInput = document.getElementById('photoInput');
  const photoPreview = document.getElementById('photoPreview');
  let photoBase64 = "";
  let photoName = "";
  let photoMimeType = "";

  photoTrigger.addEventListener('click', () => photoInput.click());

  // Resize + compress photo in-browser before upload so submit stays fast
  // on mobile data (large phone camera photos are 3-8 MB; this shrinks
  // them to under ~250 KB without losing readability).
  function compressImage(file, maxDimension, quality) {
    return new Promise((resolve, reject) => {
      const img = new Image();
      const reader = new FileReader();
      reader.onload = (e) => { img.src = e.target.result; };
      reader.onerror = reject;
      img.onload = () => {
        let { width, height } = img;
        if (width > height && width > maxDimension) {
          height = Math.round(height * (maxDimension / width));
          width = maxDimension;
        } else if (height >= width && height > maxDimension) {
          width = Math.round(width * (maxDimension / height));
          height = maxDimension;
        }
        const canvas = document.createElement('canvas');
        canvas.width = width;
        canvas.height = height;
        canvas.getContext('2d').drawImage(img, 0, 0, width, height);
        const dataUrl = canvas.toDataURL('image/jpeg', quality);
        resolve(dataUrl);
      };
      img.onerror = reject;
      reader.readAsDataURL(file);
    });
  }

  photoInput.addEventListener('change', async () => {
    const file = photoInput.files[0];
    if (!file) return;
    photoTrigger.textContent = "⏳ Processing...";
    try {
      const dataUrl = await compressImage(file, 1000, 0.7);
      photoBase64 = dataUrl.split(',')[1];
      photoName = (file.name || "driver_photo").replace(/\.[^.]+$/, "") + ".jpg";
      photoMimeType = "image/jpeg";
      photoPreview.src = dataUrl;
      photoPreview.style.display = 'inline-block';
    } catch (err) {
      statusMsg.textContent = "Photo process nahi ho payi, dobara try karo.";
      statusMsg.className = "status err";
    } finally {
      photoTrigger.textContent = "📷 Photo Capture / Upload";
    }
  });

  const form = document.getElementById('entryForm');
  const submitBtn = document.getElementById('submitBtn');
  const statusMsg = document.getElementById('statusMsg');

  function setNowInField(){
    dtField.value = new Date(Date.now() - new Date().getTimezoneOffset()*60000).toISOString().slice(0,16);
  }

  // Sends with a timeout + a couple of silent retries, since mobile
  // networks drop requests sometimes. mode:'no-cors' means we can't read
  // the response body, but a resolved fetch reliably means Google
  // received the request.
  function submitWithRetry(payload, attemptsLeft, timeoutMs){
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), timeoutMs);

    return fetch(SCRIPT_URL, {
      method: 'POST',
      mode: 'no-cors',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },
      body: JSON.stringify(payload),
      signal: controller.signal
    })
    .then(res => { clearTimeout(timer); return res; })
    .catch(err => {
      clearTimeout(timer);
      if (attemptsLeft > 1) {
        statusMsg.textContent = "Signal weak, dobara try ho raha hai...";
        statusMsg.className = "status";
        return submitWithRetry(payload, attemptsLeft - 1, timeoutMs);
      }
      throw err;
    });
  }

  let isSubmitting = false;

  form.addEventListener('submit', function(e){
    e.preventDefault();
    if (isSubmitting) return; // guards against double-tap on slow phones

    if (SCRIPT_URL.indexOf("PASTE_YOUR") === 0) {
      statusMsg.textContent = "Pehle SCRIPT_URL set karo (upar warning dekho).";
      statusMsg.className = "status err";
      return;
    }
    if (!form.checkValidity()) {
      form.reportValidity();
      return;
    }

    isSubmitting = true;
    submitBtn.disabled = true;
    submitBtn.textContent = "Submitting...";
    statusMsg.textContent = "";
    statusMsg.className = "status";

    const payload = {
      entryDateTime: document.getElementById('entryDateTime').value,
      vehicleNo: document.getElementById('vehicleNo').value.trim(),
      driverName: document.getElementById('driverName').value.trim(),
      driverMobile: document.getElementById('driverMobile').value.trim(),
      transporterName: document.getElementById('transporterName').value.trim(),
      purpose: document.getElementById('purpose').value,
      materialName: document.getElementById('materialName').value.trim(),
      quantity: document.getElementById('quantity').value.trim(),
      damageIssue: document.getElementById('damageIssue').value.trim(),
      remarks: document.getElementById('remarks').value.trim(),
      guardName: document.getElementById('guardName').value.trim(),
      photoBase64: photoBase64,
      photoName: photoName,
      photoMimeType: photoMimeType
    };

    submitWithRetry(payload, 3, 15000)
      .then(() => {
        statusMsg.textContent = "✅ Entry saved! Sheet1 me check karo.";
        statusMsg.className = "status ok";
        form.reset();
        photoPreview.style.display = 'none';
        photoBase64 = ""; photoName = ""; photoMimeType = "";
        setNowInField();
      })
      .catch(err => {
        statusMsg.textContent = "❌ Submit fail hua (" + (err.name === 'AbortError' ? 'timeout' : err.message) + "). Internet check karke dobara try karo.";
        statusMsg.className = "status err";
      })
      .finally(() => {
        isSubmitting = false;
        submitBtn.disabled = false;
        submitBtn.textContent = "Submit Entry";
      });
  });
</script>
